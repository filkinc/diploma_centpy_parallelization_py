	\begin{frame}{Математическая постановка задачи}
		
		\begin{block}{Одномерная система уравнений Эйлера}
			\[
			\partial_t \begin{bmatrix} \rho \\ \rho u \\ E \end{bmatrix} + 
			\partial_x \begin{bmatrix} \rho u \\ \rho u^2 + p \\ ( E + p)u \end{bmatrix} = 0
			\]
			
			с уравнением состояния идеального газа
			\vspace{-0.3 cm}
			\[
			p = (\gamma - 1)\left( E - \frac{1}{2}\rho u^2 \right), \quad \gamma = 1.4
			\]
		\end{block}
		
		в расчетной области $x \in [0, 1]$, с кусочно-постоянными НУ:
		\vspace{-0.15 cm}
		\[
		(\rho, u, p)|_{t=0} = 
		\begin{cases}
			(1, 0, 1) & \text{при } x \in [0, 0.3) \\
			(\alpha \cdot 1, 0, 1) & \text{при } x \in [0.3, 0.5] \\
			(1, 0, 1) & \text{при } x \in (0.5, 0.8) \\
			(2.667, -1.479, 4.5) & \text{при } x \in [0.8, 1.0]
		\end{cases}
		\]
		
		\vspace{-0.2 cm}
		где зона пониженной плотности задается с параметром $\alpha = 0.1$. \\
%		Ударная волна $(\rho_R, u_R, p_R) \approx (2.667, -1.479, 4.5)$ движется влево со скоростью $D_s \approx -2.366$. \\
		На краях области заданы граничные условия Неймана.
%		(нулевой градиент).
	\end{frame}

	% Слайд 7:
	\begin{frame}{Схема SD2}
		\begin{block}{Линейная реконструкция:}
			\begin{equation*}
				u^\mp_{i\pm1/2} = u_i \pm \frac{\Delta x}{2} \cdot (u_x)_i = u_i \pm \frac{1}{2} \text{minmod}\big( u_i - u_{i-1}, u_{i+1} - u_i \big)
			\end{equation*}
		\end{block}
		
		\vspace{- 0.2 cm}
		\begin{block}{Формула численного потока KT:}
			\begin{equation*}
				H_{i+1/2} = \frac{1}{2}\left( f(u^+_{i+1/2}) + f(u^-_{i+1/2}) - a_{i+1/2} \left( u^+_{i+1/2} - u^-_{i+1/2} \right) \right)
			\end{equation*}
			где $a_{i+1/2}$ — максимальная локальная скорость волны
		\end{block}
		
		\vspace{- 0.2 cm}
		\begin{block}{Сведение ДУЧП к системе ОДУ по времени:}
			\begin{equation*}
				\frac{d u_i}{dt} = -\frac{ H_{i+1/2} - H_{i-1/2} }{\Delta x} 
			\end{equation*}
		\end{block}
		
		\vspace{- 0.2 cm}
		\begin{block}{}
			Для интегрирование по времени используется TVD-схема Рунге-Кутты 2-го порядка (SSP-RK2).
		\end{block}
	\end{frame}
	
	\begin{frame}{Анализ сходимости: гладкое решение}
		\begin{block}{}
			Для верификации теоретического второго порядка точности схемы SD2 проведено моделирование эволюции уравнения линейного переноса.
			Порядок $p$ вычислялся методом экстраполяции Ричардсона с шагом измельчения $r=2$
		\end{block}
		
		\begin{table}[h]
			\centering
			\footnotesize
			\renewcommand{\arraystretch}{1.3}
			\begin{tabular}{lccc}
				\hline
				\textbf{Переменная} & \textbf{Ошибка (100$\to$200)} & \textbf{Ошибка (200$\to$400)} & \textbf{Порядок $p$} \\
				\hline
				u(x, t) & $8.92 \times 10^{-4}$ & $2.09 \times 10^{-4}$ & \textbf{2.09} \\
				\hline
			\end{tabular}
		\end{table}
		
		\begin{columns}[c]
			\begin{column}{0.55\textwidth}
				\[
				E = \|\mathbf{u}^{2h} - \mathbf{u}^{h}\|_{L_1} = \frac{1}{L} \sum_i |u_i^{2h} - u_i^h| \cdot \Delta x
				\]
				\[
				p = \frac{\ln(E_1 / E_2)}{\ln(r)} = \log_2 \left( \frac{|| \mathbf{u}^{4h} - \mathbf{u}^{2h} ||_{L_1}}{|| \mathbf{u}^{2h} - \mathbf{u}^{h} ||_{L_1}} \right)
				\]
			\end{column}
			\begin{column}{0.55\textwidth}
				$u$ --- скалярное решение.
				
				$L$ --- общая длина одномерной расчетной области.
				
				$\Delta x$ --- размер шага сетки.
				
				$h$ --- базовый характерный размер ячейки сетки.
			\end{column}

	\begin{frame}{Аппаратное ускорение вычислений: фреймворк JAX}
		Для преодоления вычислительных ограничений языка Python в качестве основы решателя выбран высокопроизводительный фреймворк JAX.
		
		\vspace{0.3cm}
		\textbf{Ключевые архитектурные преимущества:}
		\begin{itemize}
				\item \textbf{JIT-компиляция (Just-In-Time):} использование компилятора XLA (Accelerated Linear Algebra) для трансляции высокоуровневого кода Python в оптимизированные машинные инструкции.
				\item \textbf{Глубокая векторизация:} отказ от явных циклов по пространственной сетке; массивы обрабатываются параллельно на уровне тензорных операций.
				\item \textbf{Кроссплатформенность:} единый код (математическая логика решателя) нативно и без модификаций выполняется на многоядерных CPU и графических ускорителях (GPU/TPU).
			\end{itemize}
	\end{frame}

	\begin{frame}{Семейство численных схем Курганова-Тадмора}
		
		\begin{block}{Основные особенности метода:}
			\begin{itemize}
				\item \textbf{Вычисление потоков:} потоки на гранях ячеек определяются без вычисления матриц Якоби и характеристических переменных 
			\end{itemize}
		\end{block}
		
		\begin{block}{Подходы к дискретизации:}
			\begin{itemize}
				\item \textbf{Fully-Discrete (FD):} пространственный и временной шаги жестко связаны единым вычислительным шаблоном.
				\item \textbf{Semi-Discrete (SD):} пространственная дискретизация сводит УЧП к системе ОДУ (метод прямых). Интегрирование по времени выполняется независимыми TVD-схемами Рунге-Кутты. Различают схемы 2-го (SD2) и 3-го (SD3) порядка точности.
			\end{itemize}
		\end{block}
		
	\end{frame}
	
	\begin{frame}{Пространственная реконструкция схемы SD2}
		
		\begin{block}{TVD-свойство (Total Variation Diminishing)}
			Схема не порождает новые локальные экстремумы если
			\vspace{-0.2 cm}
			\[
			TV(u^{n+1}) \leq TV(u^{n}),
			\]
			где общая вариация решения $TV(u) = \sum_{i = 1}^{N - 1} |u_{i + 1}(t) - u_i(t)|$
			%Схема не порождает новые локальные экстремумы, то есть исключаются нефизичные осцилляции на разрывах.
		\end{block}
		
		Для обеспечения TVD-свойства используются нелинейные ограничители наклонов (limiter):
		\vspace{-0.3 cm}
		\[
		\text{minmod}(a,b) = \frac{1}{2} \big( \operatorname{sgn}(a) + \operatorname{sgn}(b) \big) \cdot \min\big( |a|, |b| \big).
		\]
	\end{frame}
