	\begin{frame}{Симейство численных схем Курганова-Тадмора}
		\begin{itemize}
			\item \textbf{Главное преимущество:} вычисление потоков без матриц Якоби и характеристических переменных.
			\item \textbf{Адаптивность ячеек:} использование суженных контрольных объемов вблизи разрывов минимизирует численную вязкость.
			\item \textbf{Два класса схем KT:}
			\begin{itemize}
				\item \textbf{FD (Fully-Discrete):} шаг по пространству и времени связан жестким шаблоном.
				\item \textbf{SD (Semi-Discrete):} пространство и время разделены, используются независимые ODE-солверы (например, Runge-Kutta).
			\end{itemize}
			\item \textbf{Порядок точности:} SD2 (линейная реконструкция) и SD3 (параболическая реконструкция).
		\end{itemize}
	\end{frame}
	
	
	% Слайд 6:
	\begin{frame}{Пространственная реконструкция схемы SD2}
		\begin{block}{Линейная реконструкция:}
			\vspace{-0.4 cm}
			\begin{equation*}
				u^\pm_{i+1/2} = u_i \pm \frac{\Delta x}{2} \cdot s_i = u_i \pm \text{minmod}\left( \frac{u_i - u_{i-1}}{\Delta x}, \frac{u_{i+1} - u_i}{\Delta x} \right)
			\end{equation*}
		\end{block}
		
		\begin{block}{TVD-свойство (Total Variation Diminishing)}
			Схема не порождает новые локальные экстремумы если
			\vspace{-0.2 cm}
			\[
			TV(u^{n+1}) \leq TV(u^{n}).
			\]
			где общая вариация решения $TV(u) = \sum_{i = 1}^{N - 1} |u_{i + 1}(t) - u_i(t)|$.
			%Схема не порождает новые локальные экстремумы, то есть исключаются нефизичные осцилляции на разрывах.
		\end{block}
		
		Для обеспечения TVD-свойства используются нелинейные ограничители наклонов (limiter): ГДЕ $\Delta x$
		\vspace{-0.3 cm}
		\[
		\text{minmod}(a,b) = \frac{1}{2} \big( \operatorname{sgn}(a) + \operatorname{sgn}(b) \big) \cdot \min\big( |a|, |b| \big).
		\]
	\end{frame}
	
	
	% Слайд 7:
	\begin{frame}{Численный поток Курганова-Тадмора и шаг по времени}
		\begin{itemize}
			\item \textbf{Сведение к системе ОДУ по времени (полудискретный вид):}
			\begin{equation*}
				\frac{d u_i}{dt} = -\frac{1}{\Delta x} \left( H_{i+1/2} - H_{i-1/2} \right)
			\end{equation*}
			
			\item \textbf{Формула численного потока KT:}
			\begin{equation*}
				H_{i+1/2} = \frac{f(u^+_{i+1/2}) + f(u^-_{i+1/2})}{2} - \frac{a_{i+1/2}}{2} \left( u^+_{i+1/2} - u^-_{i+1/2} \right)
			\end{equation*}
			
			\item $a_{i+1/2}$ — максимальная локальная скорость волны (минимальная искусственная вязкость).
			\item \textbf{Интегрирование по времени:} используется TVD-схема Рунге-Кутты 2-го порядка (SSP-RK2).
		\end{itemize}
	\end{frame}
