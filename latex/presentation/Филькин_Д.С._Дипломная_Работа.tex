	% Слайд 4: Постановка задачи
	\begin{frame}{Математическая постановка задачи}
		
		\begin{block}{Двумерная система уравнений Эйлера}
			\[
			\partial_t \begin{bmatrix} \rho \\ \rho u \\ \rho v \\ E \end{bmatrix} + 
			\partial_x \begin{bmatrix} \rho u \\ \rho u^2 + p \\ \rho u v \\ (E + p)u \end{bmatrix} + 
			\partial_y \begin{bmatrix} \rho v \\ \rho u v \\ \rho v^2 + p \\ (E + p)v \end{bmatrix} = 0
			\]
			
			с уравнением состояния
			\vspace{-0.3 cm}
			\[
				p = (\gamma - 1)\left(E - \frac{1}{2}\rho(u^2 + v^2)\right), \quad \gamma = 1.4
			\]
		\end{block}
		
		в области $(x, y) \in [0, 1] \times [0, 1]$ с НУ для задачи Римана:
		\vspace{-0.15 cm}
		\[
		(\rho, u, v, p)_{t=0} = 
		\begin{cases}
			(1.5, 0.0, 0.0, 1.5) & \text{if } x > 0.5, y > 0.5 \\
			(0.5323, 1.206, 0.0, 0.3) & \text{if } x \le 0.5, y > 0.5 \\
			(0.138, 1.206, 1.206, 0.029) & \text{if } x \le 0.5, y \le 0.5 \\
			(0.5323, 0.0, 1.206, 0.3) & \text{if } x > 0.5, y \le 0.5
		\end{cases}
		\]
		
		\vspace{-0.2 cm}
		и ГУ Дирихле, заданными начальными значениями.
	\end{frame}

	% Слайд 5:
	\begin{frame}{Семейство численных схем Курганова-Тадмора (KT)}
%		Схемы KT относятся к классу современных центральных схем высокого разрешения, свободных от необходимости точного решения задачи Римана.
		
		\begin{block}{Основные особенности метода:}
			\begin{itemize}
				\item \textbf{Вычисление потоков:} потоки на гранях ячеек определяются без вычисления матриц Якоби и характеристических переменных 
				%(в отличие от схем типа Годунова или Роэ).
				\item \textbf{Низкая диссипация:} локальная оценка максимальной скорости распространения волн (спектрального радиуса) минимизирует схемную (искусственную) вязкость.
			\end{itemize}
		\end{block}
				
		\begin{block}{Подходы к дискретизации:}
			\begin{itemize}
				\item \textbf{Fully-Discrete (FD):} пространственный и временной шаги жестко связаны единым вычислительным шаблоном.
				\item \textbf{Semi-Discrete (SD):} пространственная дискретизация сводит УЧП к системе ОДУ (метод прямых). Интегрирование по времени выполняется независимыми TVD-схемами Рунге-Кутты. Различают схемы 2-го (SD2) и 3-го (SD3) порядка точности.
			\end{itemize}
		\end{block}
		
	\end{frame}
		
	
	% Слайд 6:
	\begin{frame}{Пространственная реконструкция схемы SD2}
		\begin{block}{Линейная реконструкция:}
			\vspace{-0.4 cm}
			\begin{equation*}
				u^\mp_{i\pm1/2} = u_i \pm \frac{\Delta x}{2} \cdot (u_x)_i = u_i \pm \frac{1}{2} \text{minmod}\big( u_i - u_{i-1}, u_{i+1} - u_i \big)
			\end{equation*}
		\end{block}
		
		\begin{block}{TVD-свойство (Total Variation Diminishing)}
			Схема не порождает новые локальные экстремумы если
			\vspace{-0.2 cm}
			\[
			TV(u^{n+1}) \leq TV(u^{n}),
			\]
			где общая вариация решения $TV(u) = \sum_{i = 1}^{N - 1} |u_{i + 1}(t) - u_i(t)|$
			%Схема не порождает новые локальные экстремумы, то есть исключаются нефизичные осцилляции на разрывах.
		\end{block}
		
		Для обеспечения TVD-свойства используются нелинейные ограничители наклонов (limiter):
		\vspace{-0.3 cm}
		\[
		\text{minmod}(a,b) = \frac{1}{2} \big( \operatorname{sgn}(a) + \operatorname{sgn}(b) \big) \cdot \min\big( |a|, |b| \big).
		\]
	\end{frame}
	
	
	% Слайд 7:
	\begin{frame}{Численный поток Курганова-Тадмора и шаг по времени}
		\begin{itemize}
			\item \textbf{Сведение к системе ОДУ по времени (полудискретный вид):}
			\begin{equation*}
				\frac{d u_i}{dt} = -\frac{1}{\Delta x} \left( H_{i+1/2} - H_{i-1/2} \right)
			\end{equation*}
			
			\item \textbf{Формула численного потока KT:}
			\begin{equation*}
				H_{i+1/2} = \frac{f(u^+_{i+1/2}) + f(u^-_{i+1/2})}{2} - \frac{a_{i+1/2}}{2} \left( u^+_{i+1/2} - u^-_{i+1/2} \right)
			\end{equation*}
			
			\item $a_{i+1/2}$ — максимальная локальная скорость волны (минимальная искусственная вязкость).
			\item \textbf{Интегрирование по времени:} используется TVD-схема Рунге-Кутты 2-го порядка (SSP-RK2).
		\end{itemize}
	\end{frame}
